cmake_minimum_required(VERSION 3.15)

project(CtrLang VERSION 0.1.0 LANGUAGES C)

set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)

# Platform algılama
if(ANDROID)
    set(CTR_PLATFORM "android")
    add_definitions(-DCTR_ANDROID=1)
elseif(WIN32)
    set(CTR_PLATFORM "windows")
    add_definitions(-DCTR_WINDOWS=1)
elseif(APPLE)
    set(CTR_PLATFORM "macos")
    add_definitions(-DCTR_MACOS=1)
elseif(UNIX)
    set(CTR_PLATFORM "linux")
    add_definitions(-DCTR_LINUX=1)
endif()

message(STATUS "Platform: ${CTR_PLATFORM}")

# Kaynaklar
file(GLOB CTRLANG_SRC "src/*.c")
file(GLOB KUTUPHANE_SRC "kutuphane/*.c")

set(VENDOR_SRC
    vendor/mongoose/mongoose.c
    vendor/sqlite/sqlite3.c
    vendor/cjson/cJSON.c
    vendor/sds/sds.c
)

add_executable(ctrc
    ${CTRLANG_SRC}
    ${KUTUPHANE_SRC}
    ${VENDOR_SRC}
)

target_include_directories(ctrc PRIVATE
    src
    kutuphane
    vendor/mongoose
    vendor/sqlite
    vendor/cjson
    vendor/sds
    vendor/uthash
)

# Platform kütüphaneleri
if(CTR_PLATFORM STREQUAL "android")
    target_link_libraries(ctrc PRIVATE log)
elseif(CTR_PLATFORM STREQUAL "windows")
    target_link_libraries(ctrc PRIVATE ws2_32)
elseif(CTR_PLATFORM STREQUAL "macos")
    target_link_libraries(ctrc PRIVATE pthread)
elseif(CTR_PLATFORM STREQUAL "linux")
    target_link_libraries(ctrc PRIVATE pthread dl m)
endif()

install(TARGETS ctrc DESTINATION bin)
